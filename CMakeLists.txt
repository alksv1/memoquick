cmake_minimum_required(VERSION 3.24)
project(MemoQuick VERSION 0.2.5 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "MemoQuick uses native Windows APIs and requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(memoquick_core STATIC src/core.cpp src/storage.cpp src/markdown.cpp src/compose.cpp src/audio.cpp src/location.cpp src/geocoding.cpp src/clipboard.cpp)
target_include_directories(memoquick_core PUBLIC src PRIVATE third_party)
target_compile_definitions(memoquick_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(memoquick_core PUBLIC winhttp crypt32 shell32 ole32 advapi32 winmm windowsapp windowscodecs gdi32)
add_executable(MemoQuick WIN32 src/main.cpp src/window.cpp src/window_insert.cpp src/window_menu.cpp assets/app.rc)
target_link_libraries(MemoQuick PRIVATE memoquick_core comctl32 user32 gdi32 dwmapi imm32 psapi comdlg32)
add_executable(memoquick_tests tests/core_tests.cpp tests/clipboard_tests.cpp)
target_link_libraries(memoquick_tests PRIVATE memoquick_core)
if(MSVC)
  foreach(target memoquick_core MemoQuick memoquick_tests)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/Gy>)
  endforeach()
  target_link_options(MemoQuick PRIVATE $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
  target_link_options(MemoQuick PRIVATE /MANIFEST:NO)
endif()
enable_testing()
add_test(NAME core COMMAND memoquick_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:memoquick_tests>" "$<TARGET_FILE:MemoQuick>")
endif()
install(TARGETS MemoQuick RUNTIME DESTINATION .)
